- النتيجة: ترقية PRO → ENTERPRISE تعطي فلوس مجانية للمستخدم

### الحل المطلوب:

```solidity
function upgrade(uint256 newPlanId, address paymentToken) 
    external nonReentrant 
{
    Subscription storage sub = subscriptions[msg.sender];
    require(sub.planId < newPlanId, "Can only upgrade");
    require(sub.expiresAt > block.timestamp, "Expired, use subscribe");
    
    Plan memory oldPlan = plans[sub.planId];
    Plan memory newPlan = plans[newPlanId];
    require(newPlan.active, "New plan inactive");
    
    uint256 remaining = sub.expiresAt - block.timestamp;
    
    // حساب التكلفة المتبقية من الخطة القديمة
    uint256 oldValue = (oldPlan.priceUSDT * remaining) / oldPlan.duration;
    
    // حساب تكلفة نفس الفترة بالخطة الجديدة
    uint256 newCost = (newPlan.priceUSDT * remaining) / newPlan.duration;
    
    // ✅ الإصلاح: الحد الأدنى 25% من السعر الكامل للخطة الجديدة
    uint256 minDue = (newPlan.priceUSDT * 25) / 100;
    uint256 amountDue = newCost > minDue ? newCost : minDue;
    
    // تحويل المبلغ
    if (amountDue > 0) {
        require(
            IERC20(paymentToken).transferFrom(
                msg.sender, 
                treasury, 
                amountDue
            ), 
            "Payment failed"
        );
    }
    
    // تحديث الخطة مع إبقاء تاريخ الانتهاء الأصلي
    sub.planId = newPlanId;
    sub.totalPaid += amountDue;
    
    emit Upgraded(msg.sender, oldPlan.name, newPlan.name, amountDue);
}
